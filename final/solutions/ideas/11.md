Phương án **Greedy + Heap** (Tham lam kết hợp Hàng đợi ưu tiên) là một bước nhảy vọt về hiệu suất. Thay vì vét cạn mọi trạng thái như DP, chúng ta sẽ thực hiện các quyết định tối ưu và sử dụng Heap để "hối hận" (backtrack) khi tìm thấy lựa chọn tốt hơn.

Dưới đây là tư duy chi tiết cho cách tiếp cận này:

### 1. Chiến lược "Tham lam có hối hận"

Chúng ta sẽ ưu tiên lấp đầy **Nhóm 1** trước, sau đó tìm cách đưa các cụm vào **Nhóm 2**. Khi đưa một cụm vào Nhóm 2, có hai kịch bản xảy ra:

* **Kịch bản A:** Lấy một cụm chưa được chọn và đưa thẳng vào Nhóm 2. Trọng số tăng thêm: .
* **Kịch bản B:** Lấy một cụm **đang ở Nhóm 1** chuyển sang Nhóm 2, sau đó lấy một cụm chưa được chọn tốt nhất đưa vào Nhóm 1 để thế chỗ. Trọng số thay đổi: .

Để thực hiện việc này một cách tối ưu, chúng ta cần 3 Hàng đợi ưu tiên (Priority Queue):

1. `Q1`: Lưu  của các cụm chưa được chọn.
2. `Q2`: Lưu  của các cụm chưa được chọn.
3. `Q3`: Lưu  của các cụm **đã nằm trong Nhóm 1**.

### 2. Thuật toán cụ thể

1. **Bước 1:** Sắp xếp tất cả các cụm dữ liệu theo thứ tự  giảm dần. Chọn  cụm đầu tiên vào Nhóm 1.
* Tính tổng  của  cụm này.
* Đẩy các giá trị  của  cụm này vào `Q3`.
* Đẩy tất cả  cụm còn lại vào `Q1` (lưu ) và `Q2` (lưu ).


2. **Bước 2:** Lặp  lần để chọn  cụm vào Nhóm 2. Mỗi lần lặp, ta chọn giá trị lớn nhất giữa hai lựa chọn:
* **Lựa chọn 1:** Lấy cụm  có  lớn nhất trong số các cụm chưa chọn (từ `Q2`).
* **Lựa chọn 2:** Lấy một cụm từ Nhóm 1 chuyển sang Nhóm 2 (tăng  từ `Q3`) và bù lại bằng một cụm có  lớn nhất từ các cụm chưa chọn (từ `Q1`).
* *Lưu ý:* Cần dùng mảng đánh dấu `used[]` để biết cụm nào đã được chọn vào nhóm nào, tránh chọn trùng.



### 3. Mã nguồn minh họa (C++)

```cpp
#include <bits/stdc++.h>
using namespace std;

struct Node {
    int id, a, b;
};

bool compareA(Node x, Node y) {
    return x.a > y.a;
}

int main() {
    ios_base::sync_with_stdio(false);
    cin.tie(NULL);

    int k, n, m;
    cin >> k >> n >> m;

    vector<Node> data(k);
    for (int i = 0; i < k; i++) {
        data[i].id = i;
        cin >> data[i].a >> data[i].b;
    }

    // Sắp xếp theo a giảm dần để lấy n thằng tốt nhất vào nhóm 1 trước
    sort(data.begin(), data.end(), compareA);

    long long total_weight = 0;
    vector<bool> in_group1(k, false);
    vector<bool> used(k, false);
    
    // Q1: a_i của các cụm chưa dùng
    priority_queue<pair<int, int>> q1; 
    // Q2: b_i của các cụm chưa dùng
    priority_queue<pair<int, int>> q2; 
    // Q3: b_i - a_i của các cụm đang ở nhóm 1
    priority_queue<pair<int, int>> q3; 

    for (int i = 0; i < n; i++) {
        total_weight += data[i].a;
        in_group1[i] = true;
        used[i] = true;
        q3.push({data[i].b - data[i].a, i});
    }

    for (int i = n; i < k; i++) {
        q1.push({data[i].a, i});
        q2.push({data[i].b, i});
    }

    // Chọn m cụm vào nhóm 2
    for (int i = 0; i < m; i++) {
        // Dọn dẹp heap để lấy đúng các phần tử chưa used
        while (!q1.empty() && used[q1.top().second]) q1.pop();
        while (!q2.empty() && used[q2.top().second]) q2.pop();

        int opt1 = -2e9, opt2 = -2e9;

        if (!q2.empty()) opt1 = q2.top().first; // Cách 1: Chọn thẳng từ ngoài vào nhóm 2
        if (!q1.empty() && !q3.empty()) opt2 = q3.top().first + q1.top().first; // Cách 2: Đổi chỗ

        if (opt1 >= opt2) {
            total_weight += opt1;
            int id = q2.top().second;
            used[id] = true;
            q2.pop();
        } else {
            total_weight += opt2;
            int id_from_g1 = q3.top().second;
            int id_new_to_g1 = q1.top().second;
            
            q3.pop();
            q1.pop();

            // Cụm cũ từ nhóm 1 chuyển sang nhóm 2
            // Cụm mới vào thế chỗ ở nhóm 1
            used[id_new_to_g1] = true;
            q3.push({data[id_new_to_g1].b - data[id_new_to_g1].a, id_new_to_g1});
        }
    }

    cout << total_weight << endl;

    return 0;
}

```
