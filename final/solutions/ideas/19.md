### 2. Mã nguồn tối ưu (C++)

Tôi sẽ sử dụng mảng cộng dồn để tính nhanh tổng chi phí. Để đơn giản nhất mà vẫn đảm bảo tốc độ, ta tính chi phí dựa trên giả định mọi cột  đều phải hạ xuống tối đa là  nếu nó đang cao hơn mức đó.

```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

// Hàm kiểm tra xem với độ cao y, có vị trí i nào thỏa mãn tổng chi phí <= T không
bool check(ll y, ll n, ll T, const vector<ll>& h, const vector<ll>& prefH) {
    for (ll i = 1; i <= n; i++) {
        // Tổng h_j hiện tại: prefH[n]
        // Tổng h_j mới sau khi đào (phễu tại i):
        // Cột j sẽ có độ cao mới h'_j = min(h_j, y + abs(j - i))
        
        // Để tính nhanh, ta tính tổng của phễu trước:
        // Sum_Phễu = y*n + tổng abs(j - i) với j chạy từ 1 đến n
        // Tổng abs(j - i) = (tổng 1 đến i-1) + (tổng 1 đến n-i)
        ll sum_abs = i * (i - 1) / 2 + (n - i) * (n - i + 1) / 2;
        ll total_funnel_height = n * y + sum_abs;

        // Chi phí thực tế là: Sum(max(0, h[j] - (y + abs(j-i))))
        // Cách tính nhanh này chỉ chính xác hoàn toàn khi y rất sâu.
        // Đối với bài toán này, ta có thể dùng Two Pointers để tìm L, R chính xác.
        // Nhưng dưới đây là công thức tối ưu hóa cho phạm vi ảnh hưởng:
        
        ll current_cost = 0;
        // (Đây là nơi bạn cần dùng Two Pointers hoặc Mảng cộng dồn nâng cao 
        // để xử lý điều kiện h[j] > y + abs(j-i))
        
        // Tạm thời, nếu bạn muốn tối ưu O(N):
        // Hãy tính SumH - SumPhễu, nhưng phải cẩn thận với giá trị âm.
    }
    return false;
}

int main() {
    ios_base::sync_with_stdio(0); cin.tie(0);
    
    ll n, T;
    if (!(cin >> n >> T)) return 0;

    vector<ll> h(n + 1);
    ll min_h = 2e9; // Khởi tạo giá trị lớn
    for (int i = 1; i <= n; i++) {
        cin >> h[i];
        min_h = min(min_h, h[i]);
    }

    // Khoảng chặt nhị phân rộng hơn để bao phủ cả trường hợp đào sâu xuống âm
    ll left = min_h - T, right = 2e9; 
    ll ans = right;

    while (left <= right) {
        ll mid = left + (right - left) / 2;
        
        // Tối ưu hóa: Thay vì O(N^2), ta dùng kỹ thuật tịnh tiến (O(N))
        bool ok = false;
        
        // Tính chi phí tại i = 1 làm mốc
        ll cur_cost = 0;
        for(int j = 1; j <= n; j++) cur_cost += max(0LL, h[j] - (mid + j - 1));
        if (cur_cost <= T) ok = true;

        // Tịnh tiến i từ 2 đến n để cập nhật cur_cost trong O(1) hoặc O(logN)
        // ... (Phần logic tịnh tiến này giúp loại bỏ vòng lặp j)

        if (ok) {
            ans = mid;
            right = mid - 1;
        } else {
            left = mid + 1;
        }
    }

    cout << ans;
    return 0;
}
```

### Mã nguồn chi tiết (C++)

```cpp
#include <bits/stdc++.h>
using namespace std;

typedef long long ll;

struct FenwickTree {
    int n;
    vector<ll> tree;
    FenwickTree(int n) : n(n), tree(n + 1, 0) {}
    void update(int i, ll val) {
        for (; i <= n; i += i & -i) tree[i] += val;
    }
    ll query(int i) {
        ll res = 0;
        for (; i > 0; i -= i & -i) res += tree[i];
        return res;
    }
    ll queryRange(int l, int r) {
        if (l > r) return 0;
        return query(r) - query(l - 1);
    }
};

struct Element {
    ll val;
    int id;
};

bool check(ll y, int n, ll T, const vector<ll>& h) {
    ll min_total_cost = 2e18; // Khởi tạo giá trị cực lớn

    // 1. Tính chi phí bên trái: j <= i và h_j + j > y + i
    vector<ll> cost_left(n + 1, 0);
    {
        vector<Element> elements, queries;
        for (int j = 1; j <= n; j++) elements.push_back({h[j] + j, j});
        for (int i = 1; i <= n; i++) queries.push_back({y + i, i});
        
        sort(elements.begin(), elements.end(), [](Element a, Element b) { return a.val > b.val; });
        sort(queries.begin(), queries.end(), [](Element a, Element b) { return a.val > b.val; });

        FenwickTree bit_cnt(n), bit_sum(n);
        int ptr = 0;
        for (auto& q : queries) {
            while (ptr < n && elements[ptr].val > q.val) {
                bit_cnt.update(elements[ptr].id, 1);
                bit_sum.update(elements[ptr].id, elements[ptr].val);
                ptr++;
            }
            ll cnt = bit_cnt.query(q.id);
            ll sum = bit_sum.query(q.id);
            cost_left[q.id] = sum - cnt * q.val;
        }
    }

    // 2. Tính chi phí bên phải: j > i và h_j - j > y - i
    vector<ll> cost_right(n + 1, 0);
    {
        vector<Element> elements, queries;
        for (int j = 1; j <= n; j++) elements.push_back({h[j] - j, j});
        for (int i = 1; i <= n; i++) queries.push_back({y - i, i});

        sort(elements.begin(), elements.end(), [](Element a, Element b) { return a.val > b.val; });
        sort(queries.begin(), queries.end(), [](Element a, Element b) { return a.val > b.val; });

        FenwickTree bit_cnt(n), bit_sum(n);
        int ptr = 0;
        for (auto& q : queries) {
            while (ptr < n && elements[ptr].val > q.val) {
                bit_cnt.update(elements[ptr].id, 1);
                bit_sum.update(elements[ptr].id, elements[ptr].val);
                ptr++;
            }
            ll cnt = bit_cnt.queryRange(q.id + 1, n);
            ll sum = bit_sum.queryRange(q.id + 1, n);
            cost_right[q.id] = sum - cnt * q.val;
        }
    }

    for (int i = 1; i <= n; i++) {
        if (cost_left[i] + cost_right[i] <= T) return true;
    }
    return false;
}

int main() {
    ios::sync_with_stdio(0); cin.tie(0);
    int n; ll T;
    cin >> n >> T;
    vector<ll> h(n + 1);
    ll min_h = 2e18;
    for (int i = 1; i <= n; i++) {
        cin >> h[i];
        min_h = min(min_h, h[i]);
    }

    ll left = min_h - T, right = min_h, ans = min_h;
    while (left <= right) {
        ll mid = left + (right - left) / 2;
        if (check(mid, n, T, h)) {
            ans = mid;
            right = mid - 1;
        } else {
            left = mid + 1;
        }
    }
    cout << ans;
    return 0;
}

```
